function shiftDate(dateStr, deltaDays) {
  const d = new Date(dateStr + "T12:00:00");
  d.setDate(d.getDate() + deltaDays);
  return d.toISOString().split("T")[0];
}

function getLastNDays(baseDateStr, n = 7) {
  const result = [];
  const base = new Date(baseDateStr + "T12:00:00");
  for (let i = n - 1; i >= 0; i--) {
    const d = new Date(base);
    d.setDate(base.getDate() - i);
    const iso = d.toISOString().split("T")[0];
    const shortDay = d.toLocaleDateString("en-US", { weekday: "short" });
    const dayNum = d.getDate();
    result.push({ date: iso, label: `${shortDay} ${dayNum}`, shortDay });
  }
  return result;
}

function formatDuration(ms) {
  if (ms <= 0) return "00h 00m 00s";
  const totalMins = Math.floor(ms / 60000);
  const hrs = Math.floor(totalMins / 60);
  const mins = totalMins % 60;
  const secs = Math.floor((ms % 60000) / 1000);
  return `${String(hrs).padStart(2, "0")}h ${String(mins).padStart(2, "0")}m ${String(secs).padStart(2, "0")}s`;
}

    const entries = combo.items.map((it, idx) => ({ ...it, id: `log_${Date.now()}_${idx}`, time: now }));
    updateCurrentDay((day) => ({ ...day, foods: [...entries, ...(day.foods || [])] }));
    const totalKcal = entries.reduce((s, x) => s + x.cal, 0);
    const totalProt = entries.reduce((s, x) => s + x.prot, 0);
    triggerToast(`Logged "${combo.name}" (${totalKcal} kcal • ${totalProt}g P)!`);
  };

  const saveTodayAsCombo = () => {
    if (!currentDayData.foods || currentDayData.foods.length === 0) {
      triggerToast("Log at least 1 food today first!");
      return;
    }
    const label = comboNameInput.trim() || `Saved Day Preset (${Math.round(totals.cal)} kcal • ${Math.round(totals.prot)}g P)`;
    const newCombo = {
      id: "combo_" + Date.now(),
      name: label,
      items: currentDayData.foods.map(({ id, time, ...rest }) => rest)
    };
    setSavedCombos((prev) => [newCombo, ...prev]);
    setComboNameInput("");
    triggerToast(`Saved "${label}" to 1-Tap Presets!`);
  };

  const handleSelectFoodItem = (food) => {
    setSelectedFood(food);
    setAmountInput(food.defaultAmount || (food.isServing ? 1 : 100));
  };

  const previewNutrients = useMemo(() => {
    if (!selectedFood) return null;
    const mult = selectedFood.isServing ? (parseFloat(amountInput) || 0) : (parseFloat(amountInput) || 0) / 100;
    const scale = (val) => +((Number(val) || 0) * mult).toFixed(1);
    return {
      cal: Math.round((Number(selectedFood.cal) || 0) * mult),
      carb: scale(selectedFood.carb), prot: scale(selectedFood.prot), fat: scale(selectedFood.fat),
      fib: scale(selectedFood.fib), sug: scale(selectedFood.sug),
      sod: Math.round((Number(selectedFood.sod) || 0) * mult),
      pot: Math.round((Number(selectedFood.pot) || 0) * mult),
      calc: Math.round((Number(selectedFood.calc) || 0) * mult),
      iro: scale(selectedFood.iro), vitc: scale(selectedFood.vitc)
    };
  }, [selectedFood, amountInput]);

  const handleLogSelectedFood = () => {
    if (!selectedFood || !previewNutrients) return;
    const labelAmount = selectedFood.isServing ? `${amountInput} × ${selectedFood.unit}` : `${amountInput}g`;
    const entry = {
      id: "log_" + Date.now(),
      name: `${selectedFood.name} (${labelAmount})`,
      meal: selectedMeal,
      time: new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' }),
      ...previewNutrients
    };
    updateCurrentDay((day) => ({ ...day, foods: [entry, ...(day.foods || [])] }));
    triggerToast(`Added ${selectedFood.name} to ${selectedMeal}`);
    setActiveTab("dashboard");
  };

  const handleDeleteFoodLog = (id) => {
    updateCurrentDay((day) => ({ ...day, foods: day.foods.filter((f) => f.id !== id) }));
  };

  const handleAddWater = (ml) => {
    const now = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });
    updateCurrentDay((day) => {
      const newTotal = Math.max(0, (day.water || 0) + ml);
      const historyItem = { id: Date.now(), amount: ml, time: now };
      return { ...day, water: newTotal, waterHistory: ml > 0 ? [historyItem, ...(day.waterHistory || [])] : (day.waterHistory || []) };
    });
    if (ml > 0) triggerToast(`+${ml}ml water logged!`);
  };

  const handleAddSteps = (delta) => {
    updateCurrentDay((day) => ({ ...day, steps: Math.max(0, (day.steps || 0) + delta) }));
    if (delta > 0) triggerToast(`+${delta.toLocaleString()} steps added!`);
  };

  const handleToggleFasting = () => {
    if (fastingState.isFasting) {
      const elapsedHrs = ((Date.now() - fastingState.startTime) / 3600000).toFixed(1);
      setFastingState((prev) => ({ ...prev, isFasting: false, startTime: null }));
      triggerToast(`Fast ended after ${elapsedHrs} hours!`);
    } else {
      setFastingState((prev) => ({ ...prev, isFasting: true, startTime: Date.now() }));
      triggerToast(`Started ${fastingState.targetHours}h Intermittent Fast!`);
    }
  };

  const handleSaveCustomFood = (e) => {
    e.preventDefault();
    if (!newFood.name.trim()) return;
    const created = {
      id: "cust_" + Date.now(), name: newFood.name.trim(), category: "Custom",
      unit: newFood.isServing ? "1 Serving" : "100g", defaultAmount: newFood.isServing ? 1 : 100,
      isServing: newFood.isServing,
      cal: Number(newFood.cal) || 0, carb: Number(newFood.carb) || 0, prot: Number(newFood.prot) || 0, fat: Number(newFood.fat) || 0,
      fib: Number(newFood.fib) || 0, sug: Number(newFood.sug) || 0, sod: Number(newFood.sod) || 0, pot: Number(newFood.pot) || 0,
      calc: Number(newFood.calc) || 0, iro: Number(newFood.iro) || 0, vitc: Number(newFood.vitc) || 0
    };
    setCustomFoods((prev) => [created, ...prev]);
    setSelectedFood(created);
    setAmountInput(created.defaultAmount);
    setShowCustomModal(false);
    setNewFood({ name: "", category: "Custom", unit: "100g", isServing: false, cal: "", carb: "", prot: "", fat: "", fib: "", sug: "", sod: "", pot: "", calc: "", iro: "", vitc: "" });
    triggerToast(`Saved "${created.name}" to your Food Database!`);
  };

  const processImportedFoodJson = (rawJsonString) => {
    try {
      const parsed = JSON.parse(rawJsonString);
      let rawList = Array.isArray(parsed) ? parsed : (parsed && Array.isArray(parsed.foods)) ? parsed.foods : (parsed && typeof parsed === "object" && parsed.name) ? [parsed] : null;
      if (!rawList) {
        alert("Invalid JSON format. Expected an array of food items or an object with 'name'.");
        return;
      }
      let addedFoods = [];
      let autoLogItems = [];
      const now = new Date().toLocaleTimeString([], { hour: '2-digit', minute: '2-digit' });

      rawList.forEach((item, idx) => {
        if (!item || !item.name) return;
        const isServ = Boolean(item.isServing);
        const normalized = {
          id: item.id || `json_${Date.now()}_${idx}`,
          name: String(item.name).trim(),
          category: item.category || "JSON Import",
          unit: item.unit || (isServ ? "1 Serving" : "100g"),
          defaultAmount: Number(item.defaultAmount) || (isServ ? 1 : 100),
          isServing: isServ,
          cal: Number(item.cal ?? item.calories) || 0, carb: Number(item.carb ?? item.carbs) || 0,
          prot: Number(item.prot ?? item.protein) || 0, fat: Number(item.fat ?? item.fats) || 0,
          fib: Number(item.fib ?? item.fiber) || 0, sug: Number(item.sug ?? item.sugar) || 0,
          sod: Number(item.sod ?? item.sodium) || 0, pot: Number(item.pot ?? item.potassium) || 0,
          calc: Number(item.calc ?? item.calcium) || 0, iro: Number(item.iro ?? item.iron) || 0,
          vitc: Number(item.vitc ?? item.vitaminC) || 0
        };
        addedFoods.push(normalized);
        if (item.logNow) {
          autoLogItems.push({
            id: `log_${Date.now()}_${idx}`, name: `${normalized.name} (${normalized.unit})`,
            meal: item.meal || selectedMeal, time: now,
            cal: normalized.cal, carb: normalized.carb, prot: normalized.prot, fat: normalized.fat,
            fib: normalized.fib, sug: normalized.sug, sod: normalized.sod, pot: normalized.pot,
            calc: normalized.calc, iro: normalized.iro, vitc: normalized.vitc
          });
        }
      });

      if (addedFoods.length === 0) return;
      setCustomFoods((prev) => {
        const existingNames = new Set(addedFoods.map(f => f.name.toLowerCase()));
        return [...addedFoods, ...prev.filter(p => !existingNames.has(p.name.toLowerCase()))];
      });
      if (autoLogItems.length > 0) {
        updateCurrentDay((day) => ({ ...day, foods: [...autoLogItems, ...(day.foods || [])] }));
      }
      setSelectedFood(addedFoods[0]);
      setAmountInput(addedFoods[0].defaultAmount);
      setSelectedCategory("All");
      setShowJsonPasteModal(false);
      setJsonPasteText("");
      triggerToast(`Imported ${addedFoods.length} food(s) from JSON!`);
    } catch (err) {
      alert("Could not parse JSON. Error: " + err.message);
    }
  };

  const handleFileUploadJson = (e) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (evt) => processImportedFoodJson(evt.target.result);
    reader.readAsText(file);
    e.target.value = "";
  };

  const handleExportJSON = () => {
    const payload = { version: "NutriPulse_v3", goals, customFoods, logsByDate, weightLogs, savedCombos, exportedAt: new Date().toISOString() };
    const blob = new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" });
    const url = URL.createObjectURL(blob);
    const a = document.createElement("a");
    a.href = url;
    a.download = `NutriPulse_Backup_${selectedDate}.json`;
    a.click();
    triggerToast("Full JSON backup downloaded!");
  };

  const handleRestoreFullBackup = (e) => {
    const file = e.target.files?.[0];
    if (!file) return;
    const reader = new FileReader();
    reader.onload = (evt) => {
      try {
        const data = JSON.parse(evt.target.result);
        if (!data || (!data.logsByDate && !data.weightLogs && !data.goals)) {
          alert("Invalid NutriPulse backup file.");
          return;
        }
        if (data.goals) setGoals((prev) => ({ ...prev, ...data.goals }));
        if (Array.isArray(data.customFoods)) setCustomFoods(data.customFoods);
        if (data.logsByDate && typeof data.logsByDate === "object") setLogsByDate(data.logsByDate);
        if (Array.isArray(data.weightLogs)) setWeightLogs(data.weightLogs);
        if (Array.isArray(data.savedCombos)) setSavedCombos(data.savedCombos);
        triggerToast("Full NutriPulse backup restored!");
      } catch (err) {
        alert("Failed to restore backup: " + err.message);
      }
    };
    reader.readAsText(file);
    e.target.value = "";
  };

  const handleAddOrUpdateWeight = (e) => {
    e.preventDefault();
    const val = parseFloat(weightInputValue);
    if (!val || val <= 0) return;
    setWeightLogs((prev) => {
      const updated = [...prev.filter((w) => w.date !== weightInputDate), { date: weightInputDate, weight: +val.toFixed(1), note: weightInputNote.trim() }];
      updated.sort((a, b) => a.date.localeCompare(b.date));
      return updated;
    });
    setWeightInputValue("");
    setWeightInputNote("");
    triggerToast(`Saved weight: ${val.toFixed(1)} kg on ${weightInputDate}`);
  };

  const handleDeleteWeight = (dateKey) => setWeightLogs((prev) => prev.filter((w) => w.date !== dateKey));

  const filteredWeightLogs = useMemo(() => {
    const sorted = [...weightLogs].sort((a, b) => a.date.localeCompare(b.date));
    if (weightRangeDays === 365 || sorted.length === 0) return sorted;
    const cutoff = new Date(selectedDate + "T12:00:00");
    cutoff.setDate(cutoff.getDate() - weightRangeDays);
    const cutoffStr = cutoff.toISOString().split("T")[0];
    const within = sorted.filter((w) => w.date >= cutoffStr);
    return within.length > 0 ? within : sorted.slice(-Math.min(sorted.length, 7));
  }, [weightLogs, weightRangeDays, selectedDate]);

  const weightStats = useMemo(() => {
    if (weightLogs.length === 0) return { start: 0, current: 0, change: 0, remaining: 0, bmi: 0 };
    const sorted = [...weightLogs].sort((a, b) => a.date.localeCompare(b.date));
    const start = sorted[0].weight;
    const current = sorted[sorted.length - 1].weight;
    const change = +(current - start).toFixed(1);
    const remaining = +(current - (goals.targetWeight || 110)).toFixed(1);
    const hMeters = (goals.heightCm || 181) / 100;
    const bmi = +(current / (hMeters * hMeters)).toFixed(1);
    return { start, current, change, remaining, bmi };
  }, [weightLogs, goals.targetWeight, goals.heightCm]);

  const weeklyDaysData = useMemo(() => {
    return getLastNDays(selectedDate, 7).map((d) => {
      const dayLog = logsByDate[d.date] || { water: 0, steps: 0, foods: [] };
      const dayCal = (dayLog.foods || []).reduce((sum, item) => sum + (Number(item.cal) || 0), 0);
      const dayProt = (dayLog.foods || []).reduce((sum, item) => sum + (Number(item.prot) || 0), 0);
      return { ...d, cal: Math.round(dayCal), prot: Math.round(dayProt), water: Math.round(dayLog.water || 0), steps: Math.round(dayLog.steps || 0) };
    });
  }, [logsByDate, selectedDate]);

  const activeStreak = useMemo(() => {
    let streak = 0;
    let checkDate = todayKey;
    for (let i = 0; i < 60; i++) {
      const entry = logsByDate[checkDate];
      if (entry && ((entry.foods && entry.foods.length > 0) || (entry.steps || 0) >= 8000)) {
        streak++;
        checkDate = shiftDate(checkDate, -1);
      } else break;
    }
    return streak;
  }, [logsByDate, todayKey]);

  const filteredFoods = useMemo(() => {
    return allFoods.filter((f) => {
      const matchesCat = selectedCategory === "All" || f.category === selectedCategory;
      const matchesSearch = f.name.toLowerCase().includes(searchQuery.toLowerCase());
      return matchesCat && matchesSearch;
    });
  }, [allFoods, selectedCategory, searchQuery]);

  const calPct = Math.min(100, Math.round((totals.cal / (goals.cal || 1)) * 100));
  const calRemaining = Math.max(0, (goals.cal || 0) - Math.round(totals.cal));
  const ringRadius = 56;
  const ringCircumference = 2 * Math.PI * ringRadius;
  const ringOffset = ringCircumference - (calPct / 100) * ringCircumference;
  const waterPct = Math.min(100, Math.round(((currentDayData.water || 0) / (goals.water || 1)) * 100));
  const stepsPct = Math.min(100, Math.round(((currentDayData.steps || 0) / (goals.steps || 10000)) * 100));
  const estStepKcal = Math.round((currentDayData.steps || 0) * 0.055);

  const fastingElapsedMs = fastingState.isFasting && fastingState.startTime ? Math.max(0, nowTick - fastingState.startTime) : 0;
  const fastingTargetMs = (fastingState.targetHours || 16) * 3600000;
  const fastingPct = Math.min(100, Math.round((fastingElapsedMs / fastingTargetMs) * 100));

  return (
    <div className="min-h-screen max-w-md mx-auto bg-[#090d16] text-slate-100 flex flex-col pb-24 relative select-none">
      {toastMsg && (
        <div className="fixed top-4 left-1/2 -translate-x-1/2 z-50 bg-emerald-500 text-slate-950 font-bold text-xs px-4 py-2.5 rounded-full shadow-xl border border-emerald-300 animate-bounce text-center max-w-[90vw]">
          {toastMsg}
        </div>
      )}

      <header className="sticky top-0 z-30 bg-[#090d16]/95 backdrop-blur-md border-b border-slate-800/80 px-3.5 py-2.5 flex items-center justify-between gap-2">
        <div className="flex items-center gap-2 min-w-0">
          <div className="w-8 h-8 rounded-xl bg-gradient-to-br from-emerald-400 to-cyan-500 flex items-center justify-center text-slate-950 font-black text-base shadow-md shrink-0">N</div>
          <div className="truncate">
            <div className="flex items-center gap-1.5">
              <h1 className="text-sm font-extrabold tracking-tight bg-gradient-to-r from-emerald-400 via-teal-300 to-cyan-400 bg-clip-text text-transparent">NutriPulse v3</h1>
              {activeStreak > 0 && (
                <span className="text-[10px] font-bold bg-amber-500/20 text-amber-300 border border-amber-500/30 px-1.5 py-0.5 rounded-full">🔥 {activeStreak}d</span>
              )}
            </div>
            <p className="text-[10px] text-slate-400 truncate">Macros • Fasting • Steps • Weight</p>
          </div>
        </div>

        <div className="flex items-center gap-1 shrink-0">
          <button onClick={() => setSelectedDate(shiftDate(selectedDate, -1))} className="w-7 h-7 rounded-lg bg-slate-900 border border-slate-700/80 text-slate-300 text-xs font-bold flex items-center justify-center active:scale-90">‹</button>
          <input type="date" value={selectedDate} onChange={(e) => setSelectedDate(e.target.value)} className="bg-slate-900 border border-slate-700/80 text-[11px] text-slate-200 rounded-lg px-2 py-1 focus:outline-none focus:border-emerald-500" />
          <button onClick={() => setSelectedDate(shiftDate(selectedDate, 1))} className="w-7 h-7 rounded-lg bg-slate-900 border border-slate-700/80 text-slate-300 text-xs font-bold flex items-center justify-center active:scale-90">›</button>
        </div>
      </header>

      <main className="flex-1 p-3.5 space-y-3.5">
        {activeTab === "dashboard" && (
          <>
            <section className="bg-gradient-to-r from-amber-950/50 via-slate-900 to-slate-900 border border-amber-500/30 rounded-3xl p-3.5 shadow-xl">
              <div className="flex items-center justify-between gap-2">
                <div>
                  <h2 className="text-xs font-extrabold text-amber-300 uppercase tracking-wider">⏱️ Intermittent Fasting ({fastingState.targetHours}:8)</h2>
                  {fastingState.isFasting ? (
                    <div className="mt-1">
                      <span className="text-lg font-black text-white font-mono">{formatDuration(fastingElapsedMs)}</span>
                      <span className="text-[11px] text-amber-300/90 ml-2 font-semibold">({fastingPct}% of {fastingState.targetHours}h)</span>
                    </div>
                  ) : (
                    <p className="text-[11px] text-slate-400 mt-0.5">Eating window open • Tap Start after your last meal</p>
                  )}
                </div>
                <button onClick={handleToggleFasting} className={`shrink-0 px-3.5 py-2 rounded-xl text-xs font-black transition active:scale-95 ${fastingState.isFasting ? "bg-rose-500/20 border border-rose-500/50 text-rose-300" : "bg-amber-400 text-slate-950 shadow-md"}`}>
                  {fastingState.isFasting ? "End Fast" : "Start Fast"}
                </button>
              </div>
              {fastingState.isFasting && (
                <div className="w-full h-1.5 bg-slate-950 rounded-full overflow-hidden mt-2.5">
                  <div className="bg-gradient-to-r from-amber-500 to-emerald-400 h-full rounded-full transition-all duration-500" style={{ width: `${fastingPct}%` }} />
                </div>
              )}
            </section>

            <section className="bg-slate-900/90 border border-slate-800 rounded-3xl p-4 shadow-xl">
              <div className="flex items-center justify-between gap-4">
                <div className="relative w-32 h-32 flex items-center justify-center shrink-0">
                  <svg className="w-full h-full -rotate-90" viewBox="0 0 136 136">
                    <circle cx="68" cy="68" r={ringRadius} stroke="#1e293b" strokeWidth="11" fill="transparent" />
                    <circle cx="68" cy="68" r={ringRadius} stroke="url(#gradRing)" strokeWidth="11" strokeDasharray={ringCircumference} strokeDashoffset={ringOffset} strokeLinecap="round" fill="transparent" className="transition-all duration-500" />
                    <defs>
                      <linearGradient id="gradRing" x1="0%" y1="0%" x2="100%" y2="100%">
                        <stop offset="0%" stopColor="#10b981" />
                        <stop offset="100%" stopColor="#06b6d4" />
                      </linearGradient>
                    </defs>
                  </svg>
                  <div className="absolute inset-0 flex flex-col items-center justify-center text-center">
                    <span className="text-[9px] uppercase tracking-wider text-slate-400 font-semibold">Remaining</span>
                    <span className="text-xl font-black text-white leading-tight">{calRemaining}</span>
                    <span className="text-[10px] text-emerald-400 font-medium">{Math.round(totals.cal)} / {goals.cal}</span>
                  </div>
                </div>

                <div className="flex-1 space-y-2.5">
                  <div>
                    <div className="flex justify-between text-xs mb-1">
                      <span className="font-bold text-rose-400">Protein</span>
                      <span className="text-slate-300 font-semibold">{Math.round(totals.prot)} / {goals.prot}g</span>
                    </div>
                    <div className="w-full h-2 bg-slate-800 rounded-full overflow-hidden">
                      <div className="bg-gradient-to-r from-rose-500 to-pink-400 h-full rounded-full transition-all duration-300" style={{ width: `${Math.min(100, (totals.prot / (goals.prot || 1)) * 100)}%` }} />
                    </div>
                  </div>
                  <div>
                    <div className="flex justify-between text-xs mb-1">
                      <span className="font-bold text-amber-400">Carbs</span>
                      <span className="text-slate-300 font-semibold">{Math.round(totals.carb)} / {goals.carb}g</span>
                    </div>
                    <div className="w-full h-2 bg-slate-800 rounded-full overflow-hidden">
                      <div className="bg-gradient-to-r from-amber-500 to-yellow-400 h-full rounded-full transition-all duration-300" style={{ width: `${Math.min(100, (totals.carb / (goals.carb || 1)) * 100)}%` }} />
                    </div>
                  </div>
                  <div>
                    <div className="flex justify-between text-xs mb-1">
                      <span className="font-bold text-purple-400">Fat</span>
                      <span className="text-slate-300 font-semibold">{Math.round(totals.fat)} / {goals.fat}g</span>
                    </div>
                    <div className="w-full h-2 bg-slate-800 rounded-full overflow-hidden">
                      <div className="bg-gradient-to-r from-purple-500 to-indigo-400 h-full rounded-full transition-all duration-300" style={{ width: `${Math.min(100, (totals.fat / (goals.fat || 1)) * 100)}%` }} />
                    </div>
                  </div>
                </div>
              </div>

              {savedCombos.length > 0 && (
                <div className="mt-3.5 pt-3 border-t border-slate-800/90 space-y-2">
                  <span className="text-[10px] uppercase tracking-wider text-slate-400 font-bold block">⚡ 1-Tap Meal Presets</span>
                  <div className="flex gap-2 overflow-x-auto no-scrollbar pb-0.5">
                    {savedCombos.map((c) => (
                      <button key={c.id} onClick={() => applySavedCombo(c)} className="shrink-0 bg-emerald-500/15 hover:bg-emerald-500/25 border border-emerald-500/40 text-emerald-300 text-[11px] font-bold px-3 py-1.5 rounded-xl active:scale-95 transition">
                        + {c.name}
                      </button>
                    ))}
                  </div>
                </div>
              )}
            </section>

            <section className="bg-gradient-to-br from-emerald-950/60 via-slate-900 to-slate-900 border border-emerald-800/50 rounded-3xl p-4 shadow-xl">
              <div className="flex items-center justify-between mb-2">
                <div>
                  <h2 className="text-sm font-bold text-emerald-300">👟 Daily Step Counter</h2>
                  <p className="text-[11px] text-slate-400">Goal: {(goals.steps || 10000).toLocaleString()} • ~{estStepKcal} kcal burned</p>
                </div>
                <div className="text-right">
                  <span className="text-xl font-black text-emerald-400">{(currentDayData.steps || 0).toLocaleString()}</span>
                  <span className="block text-[10px] text-emerald-200">{stepsPct}% done</span>
                </div>
              </div>
              <div className="w-full h-2.5 bg-slate-950 rounded-full overflow-hidden mb-3 border border-emerald-950">
                <div className="bg-gradient-to-r from-emerald-500 to-lime-400 h-full rounded-full transition-all duration-500" style={{ width: `${stepsPct}%` }} />
              </div>
              <div className="grid grid-cols-4 gap-1.5">
                <button onClick={() => handleAddSteps(1000)} className="bg-emerald-500/20 border border-emerald-500/40 text-emerald-200 py-1.5 rounded-xl text-xs font-bold active:scale-95">+1,000</button>

              <div className="bg-slate-900/90 border border-slate-800 p-2.5 rounded-2xl text-center"><span className="text-[9px] uppercase text-slate-400 block">Current</span><strong className="text-base font-black text-white">{weightStats.current}kg</strong></div>
              <div className="bg-slate-900/90 border border-slate-800 p-2.5 rounded-2xl text-center"><span className="text-[9px] uppercase text-slate-400 block">Change</span><strong className={`text-base font-black ${weightStats.change <= 0 ? "text-emerald-400" : "text-rose-400"}`}>{weightStats.change > 0 ? `+${weightStats.change}` : weightStats.change}kg</strong></div>
              <div className="bg-slate-900/90 border border-slate-800 p-2.5 rounded-2xl text-center"><span className="text-[9px] uppercase text-slate-400 block">To {goals.targetWeight}kg</span><strong className="text-base font-black text-cyan-400">{weightStats.remaining > 0 ? `-${weightStats.remaining}` : "Goal!"}</strong></div>
              <div className="bg-slate-900/90 border border-slate-800 p-2.5 rounded-2xl text-center"><span className="text-[9px] uppercase text-slate-400 block">BMI</span><strong className="text-base font-black text-amber-400">{weightStats.bmi}</strong></div>
            </section>

            <section className="bg-slate-900/90 border border-slate-800 rounded-3xl p-4 shadow-xl">
              <div className="flex items-center justify-between mb-3">
                <h2 className="text-sm font-extrabold text-white">⚖️ Weight Trend Graph</h2>
                <div className="flex gap-1">
                  {[{ label: "7D", days: 7 }, { label: "30D", days: 30 }, { label: "90D", days: 90 }, { label: "All", days: 365 }].map((r) => (
                    <button key={r.days} onClick={() => setWeightRangeDays(r.days)} className={`px-2 py-1 rounded-lg text-[10px] font-bold border ${weightRangeDays === r.days ? "bg-emerald-500 text-slate-950 border-emerald-400" : "bg-slate-950 text-slate-400 border-slate-800"}`}>{r.label}</button>
                  ))}
                </div>
              </div>
              {filteredWeightLogs.length > 0 && (() => {
                const weights = filteredWeightLogs.map((w) => w.weight);
                const minW = Math.floor(Math.min(...weights, goals.targetWeight || 110) - 2);
                const maxW = Math.ceil(Math.max(...weights) + 2);
                const spanW = Math.max(4, maxW - minW);
                const svgW = 340, svgH = 170, padL = 36, padR = 18, padT = 18, padB = 32;
                const plotW = svgW - padL - padR, plotH = svgH - padT - padB;
                const points = filteredWeightLogs.map((entry, idx) => ({
                  x: filteredWeightLogs.length === 1 ? padL + plotW / 2 : padL + (idx / (filteredWeightLogs.length - 1)) * plotW,
                  y: padT + plotH - ((entry.weight - minW) / spanW) * plotH,
                  ...entry
                }));
                const targetY = padT + plotH - (((goals.targetWeight || 110) - minW) / spanW) * plotH;
                const polylineStr = points.map((p) => `${p.x},${p.y}`).join(" ");
                return (
                  <svg viewBox={`0 0 ${svgW} ${svgH}`} className="w-full h-52">
                    <line x1={padL} y1={targetY} x2={svgW - padR} y2={targetY} stroke="#06b6d4" strokeDasharray="4 2" strokeWidth="1.2" />
                    {points.length > 1 && <polyline fill="none" stroke="#10b981" strokeWidth="2.5" points={polylineStr} />}
                    {points.map((p) => (
                      <g key={p.date}>
                        <circle cx={p.x} cy={p.y} r="4" fill="#090d16" stroke="#34d399" strokeWidth="2.5" />
                        <text x={p.x} y={p.y - 7} textAnchor="middle" fontSize="8" fill="#ecfdf5" fontWeight="bold">{p.weight}</text>
                        <text x={p.x} y={svgH - 10} textAnchor="middle" fontSize="7.5" fill="#94a3b8">{p.date.slice(5)}</text>
                      </g>
                    ))}
                  </svg>
                );
              })()}
            </section>

            <section className="bg-slate-900/90 border border-slate-800 rounded-3xl p-4 shadow-xl">
              <h3 className="text-sm font-bold text-emerald-400 mb-2.5">➕ Log Body Weight</h3>
              <form onSubmit={handleAddOrUpdateWeight} className="space-y-2.5">
                <div className="grid grid-cols-2 gap-2.5">
                  <input type="date" required value={weightInputDate} onChange={(e) => setWeightInputDate(e.target.value)} className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3 py-2 text-xs text-white" />
                  <input type="number" step="0.1" required placeholder="Weight (kg)" value={weightInputValue} onChange={(e) => setWeightInputValue(e.target.value)} className="w-full bg-slate-950 border border-emerald-500/60 rounded-xl px-3 py-2 text-xs font-bold text-white" />
                </div>
                <input type="text" placeholder="Optional note" value={weightInputNote} onChange={(e) => setWeightInputNote(e.target.value)} className="w-full bg-slate-950 border border-slate-700 rounded-xl px-3 py-2 text-xs text-white" />
                <button type="submit" className="w-full bg-gradient-to-r from-emerald-500 to-teal-400 text-slate-950 font-black py-2.5 rounded-xl text-xs shadow-lg">Save Weight Entry</button>
              </form>
            </section>

            <section className="bg-slate-900/90 border border-slate-800 rounded-3xl p-4 space-y-2">
              <h3 className="text-xs font-bold uppercase text-slate-400">History ({weightLogs.length})</h3>
              {weightLogs.slice().sort((a, b) => b.date.localeCompare(a.date)).map((w) => (
                <div key={w.date} className="bg-slate-950 border border-slate-800 px-3 py-2.5 rounded-2xl flex items-center justify-between text-xs">
                  <div><span className="font-bold text-white">{w.weight} kg</span> <span className="text-slate-400 ml-2">{w.date}</span></div>
                  <button onClick={() => handleDeleteWeight(w.date)} className="text-rose-400 px-2 py-1 font-bold">✕</button>
                </div>
              ))}
            </section>
          </div>
        )}

        {activeTab === "settings" && (
          <div className="space-y-4">
            <section className="bg-slate-900/90 border border-slate-800 rounded-3xl p-4 space-y-3">
              <h2 className="text-sm font-bold text-emerald-400">🎯 Daily Targets, Fasting & Body Profile</h2>
              <div className="grid grid-cols-2 gap-2.5 text-xs">
                {[
                  { key: "cal", label: "Daily Calories (kcal)" }, { key: "prot", label: "Protein Target (g)" },
                  { key: "carb", label: "Carbs Target (g)" }, { key: "fat", label: "Fat Target (g)" },
                  { key: "water", label: "Water Goal (ml)" }, { key: "steps", label: "Daily Steps Goal" },
                  { key: "fastingHours", label: "Fasting Window (hrs)" }, { key: "targetWeight", label: "Target Weight (kg)" },
                  { key: "heightCm", label: "Height (cm)" }, { key: "fib", label: "Fiber Target (g)" },
                  { key: "sod", label: "Sodium Limit (mg)" }, { key: "pot", label: "Potassium Goal (mg)" }
                ].map((g) => (
                  <div key={g.key} className="bg-slate-950 border border-slate-800 p-2.5 rounded-xl">
                    <label className="text-[11px] text-slate-400 block mb-1">{g.label}</label>
                    <input type="number" value={goals[g.key]} onChange={(e) => { const val = Number(e.target.value) || 0; setGoals({ ...goals, [g.key]: val }); if (g.key === "fastingHours") setFastingState((p) => ({ ...p, targetHours: val || 16 })); }} className="w-full bg-slate-900 border border-slate-700 rounded-lg px-2.5 py-1.5 text-xs font-bold text-white" />
                  </div>
                ))}
              </div>
            </section>

            <section className="bg-slate-900/90 border border-slate-800 rounded-3xl p-4 space-y-2.5">
              <h2 className="text-sm font-bold text-slate-200">💾 Full App Backup, Restore & Reset</h2>
              <input ref={backupImportRef} type="file" accept=".json,application/json" onChange={handleRestoreFullBackup} className="hidden" />
              <div className="grid grid-cols-2 gap-2">
                <button onClick={handleExportJSON} className="bg-emerald-500/20 border border-emerald-500/40 text-emerald-300 font-bold py-2.5 rounded-xl text-xs">⬇ Export Full Backup</button>
                <button onClick={() => backupImportRef.current?.click()} className="bg-indigo-500/20 border border-indigo-500/40 text-indigo-300 font-bold py-2.5 rounded-xl text-xs">⬆ Restore Full Backup</button>
              </div>
              <button onClick={() => { if (confirm(`Clear all logged data for ${selectedDate}?`)) { updateCurrentDay(() => ({ water: 0, steps: 0, waterHistory: [], foods: [] })); triggerToast("Cleared selected day's log."); } }} className="w-full bg-rose-500/20 border border-rose-500/40 text-rose-300 font-bold py-2.5 rounded-xl text-xs">🗑 Reset Selected Day ({selectedDate})</button>
            </section>
          </div>
        )}
      </main>

      <nav className="fixed bottom-0 left-0 right-0 max-w-md mx-auto bg-[#090d16]/95 backdrop-blur-lg border-t border-slate-800/90 px-2 py-2 flex justify-around items-center z-40">
        {[
          { id: "dashboard", icon: "📊", label: "Daily" },
          { id: "logger", icon: "🍽️", label: "Foods/JSON" },
          { id: "weekly", icon: "📈", label: "Weekly" },
          { id: "weight", icon: "⚖️", label: "Weight" },
          { id: "settings", icon: "⚙️", label: "Goals" }
        ].map((tab) => (
          <button key={tab.id} onClick={() => setActiveTab(tab.id)} className={`flex flex-col items-center py-1 px-2.5 rounded-2xl transition ${activeTab === tab.id ? "text-emerald-400 bg-emerald-500/10 font-bold" : "text-slate-400"}`}>
            <span className="text-base">{tab.icon}</span>
            <span className="text-[10px] mt-0.5">{tab.label}</span>
          </button>
        ))}
      </nav>
    </div>
  );
}
